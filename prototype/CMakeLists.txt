cmake_minimum_required(VERSION 3.21)
project(cpmimage_review VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets Test)
add_executable(cpmimage-review src/main.cpp src/window.cpp src/window.h)
target_link_libraries(cpmimage-review PRIVATE Qt6::Widgets)
if(APPLE)
  set_target_properties(cpmimage-review PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER org.cpmarchives.cpmimage.review)
endif()
if(WIN32)
  set_target_properties(cpmimage-review PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(gui-check tests/gui_check.cpp src/window.cpp src/window.h)
  target_include_directories(gui-check PRIVATE src)
  target_link_libraries(gui-check PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME gui-check COMMAND gui-check)
  set_tests_properties(gui-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
