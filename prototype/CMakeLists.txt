cmake_minimum_required(VERSION 3.21)
project(cpmimage_review VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets Test)
add_executable(cpmimage-review src/main.cpp src/window.cpp src/window.h src/filename_policy.cpp src/filename_policy.h resources/icons.qrc)
target_link_libraries(cpmimage-review PRIVATE Qt6::Widgets)
if(APPLE)
  target_sources(cpmimage-review PRIVATE resources/cpmimage.icns)
  set_source_files_properties(resources/cpmimage.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(cpmimage-review PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_GUI_IDENTIFIER org.cpmarchives.cpmimage.review
    MACOSX_BUNDLE_ICON_FILE cpmimage.icns)
endif()
if(WIN32)
  enable_language(RC)
  target_sources(cpmimage-review PRIVATE resources/cpmimage.rc)
  target_include_directories(cpmimage-review PRIVATE resources)
  set_target_properties(cpmimage-review PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(gui-check tests/gui_check.cpp src/window.cpp src/window.h src/filename_policy.cpp src/filename_policy.h)
  target_include_directories(gui-check PRIVATE src)
  target_link_libraries(gui-check PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME gui-check COMMAND gui-check)
  set_tests_properties(gui-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
