cmake_minimum_required(VERSION 3.25)
project(cpmwb_imagedisk LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory("${LIBIMD_SOURCE}" libimd)
add_compile_definitions(CMAKE_VERSION_STR="0.1.1" GIT_VERSION_STR="294e9af")
if(MSVC)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

foreach(tool imda imdchk imdu imdcmp bin2imd imdv)
  add_executable(${tool} "${IMD_UTILS_SOURCE}/src/${tool}.c")
  target_link_libraries(${tool} PRIVATE libimd)
  install(TARGETS ${tool} RUNTIME DESTINATION bin)
endforeach()
target_link_libraries(imdchk PRIVATE libimdchk)
target_link_libraries(imdv PRIVATE libimdf)
if(WIN32)
  file(GLOB console_sources "${PDCURSES_SOURCE}/pdcurses/*.c" "${PDCURSES_SOURCE}/wincon/*.c")
  add_library(imdv_curses STATIC ${console_sources})
  target_include_directories(imdv_curses PUBLIC "${PDCURSES_SOURCE}")
  target_link_libraries(imdv_curses PUBLIC winmm)
  target_link_libraries(imdv PRIVATE imdv_curses)
else()
  find_package(Curses REQUIRED)
  target_include_directories(imdv PRIVATE ${CURSES_INCLUDE_DIRS})
  target_link_libraries(imdv PRIVATE ${CURSES_LIBRARIES})
endif()
